document.getElementById("surprise").textContent = surprises[random]; } const questions = [ {q:"Your perfect weekend?",a:["Exploring","Staying home","Meeting friends","Creating art"]}, {q:"Choose a magical power.",a:["Teleportation","Time travel","Invisibility","Mind reading"]}, {q:"Pick a colour.",a:["Purple","Blue","Black","Pink"]}, {q:"What describes you best?",a:["Curious","Calm","Adventurous","Creative"]}, {q:"Choose a destination.",a:["Outer space","Beach","Castle","Colourful city"]}, {q:"Favourite activity?",a:["Puzzles","Music","Travelling","Art"]}, {q:"Choose a companion.",a:["Robot","Dragon","Talking cat","Magical bird"]}, {q:"Choose a symbol.",a:["⭐","🌙","🔥","🌈"]}, {q:"Your dream discovery?",a:["Secret door","Hidden treasure","New planet","Lost city"]}, {q:"Pick a fictional world.",a:["Future city","Enchanted forest","Underwater kingdom","Space station"]} ]; let current = 0; let answers = []; let mode = "quick"; let infinitePool = []; let lastQuestion = -1; function shuffle(array) { return array.sort(() => Math.random() - 0.5); } function startQuickQuiz() { mode = "quick"; current = 0; answers = []; document.getElementById("quiz").style.display = "block"; document.getElementById("result").style.display = "none"; showQuestion(); } function startInfiniteQuiz() { mode = "infinite"; current = 0; answers = []; infinitePool = shuffle([...Array(questions.length).keys()]); lastQuestion = -1; document.getElementById("quiz").style.display = "block"; document.getElementById("result").style.display = "none"; showQuestion(); } function showQuestion() { if (mode === "infinite" && infinitePool.length === 0) { infinitePool = shuffle([...Array(questions.length).keys()]); if (infinitePool[0] === lastQuestion) { [infinitePool[0], infinitePool[1]] = [infinitePool[1], infinitePool[0]]; } } let questionIndex; if (mode === "quick") { questionIndex = current; } else { questionIndex = infinitePool.shift(); lastQuestion = questionIndex; } const q = questions[questionIndex]; document.getElementById("question").textContent = q.q; document.getElementById("progress").textContent = mode === "quick" ? `Question ${current + 1} of 10` : `Infinite Mode ♾️ | Question ${current + 1}`; const box = document.getElementById("options"); box.innerHTML = ""; q.a.forEach((answer, index) => { const btn = document.createElement("button"); btn.className = "option"; btn.textContent = answer; btn.onclick = function() { answers.push(index); current++; if (mode === "quick" && current >= 10) { showResult(); } else { showQuestion(); } }; box.appendChild(btn); }); if (mode === "infinite") { const finish = document.createElement("button"); finish.textContent = "Finish Quiz"; finish.onclick = showResult; box.appendChild(finish); } } function showResult() { document.getElementById("quiz").style.display = "none"; document.getElementById("result").style.display = "block"; const types = [ ["THE EXPLORER 🌍","You love discovering new things!"], ["THE DREAMER 🌙","Your imagination is full of possibilities!"], ["THE ADVENTURER 🔥","You bring curiosity and excitement!"], ["THE CREATOR 🎨","You love originality and creativity!"] ]; const counts = [0,0,0,0]; answers.forEach(answer => counts[answer]++); let winner = counts.indexOf(Math.max(...counts)); document.getElementById("resultTitle").textContent = types[winner][0]; document.getElementById("resultText").textContent = types[winner][1] + ` You answered ${answers.length} questions!`; }
